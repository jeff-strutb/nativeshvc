cmake_minimum_required(VERSION 3.20)
project(zelda3 C)
add_subdirectory(${NSHVC_ROOT} nshvc)
include(${NSHVC_ROOT}/cmake/NshvcGame.cmake)
file(GLOB game_sources src/*.c)
if(DEFINED ENV{ZELDA3_CHECK})
  list(FILTER game_sources EXCLUDE REGEX "/game\\.c$")
  list(APPEND game_sources tests/check.c)
endif()
nshvc_game(zelda3 SOURCES ${game_sources} ASSETS data TITLE "The Legend of Zelda: A Link to the Past")
target_include_directories(zelda3 PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
# quote includes only: src/features.h must not hide the C library's
target_compile_options(zelda3 PRIVATE -iquote ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_options(zelda3 PRIVATE -w)
